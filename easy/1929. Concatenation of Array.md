i started an (ch:num) loop and stored ch at ans[i] and ans[i+n]   
where n is the size of num

```
class Solution {
public:
    vector<int> getConcatenation(vector<int>& nums) {
        int n=nums.size();
        vector<int> ans(2*n);
        int i=0;
        for(auto& ch:nums){

            ans[i]=ch;
            ans[i+n]=ch;
            i++;
        }
        
        return ans;
    }
};
```
